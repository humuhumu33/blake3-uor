  name: hash_is_its_trace_digest,
  parameters: [message : Bytes, outLen : Nat],
  statement: ((Blake3::hash message outLen) = (Blake3::digestOf (Blake3::finalTrace (Blake3::update (Blake3::hasherWith (Blake3::iv) (0 : UInt32)) message) outLen))),
  proof: reflexivity{},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
