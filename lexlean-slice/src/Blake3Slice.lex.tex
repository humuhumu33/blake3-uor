definition{
  name: chunkCount,
  parameters: [len : Nat],
  result: Nat,
  body: (if (len =? 0) then 1 else ((prim quotient (prim subtract len 1 : Nat) 1024 0 : Nat) + 1)),
  executable: true
}

definition{
  name: chunkSize,
  parameters: [len : Nat, chunk : Nat],
  result: Nat,
  body: (let begin : Nat := (prim multiply chunk 1024 : Nat); (if ((begin + 1024) <=? len) then 1024 else (prim subtract len begin : Nat))),
  executable: true
}

definition{
  name: leftSpan,
  parameters: [size : Nat],
  result: Nat,
  body: (first (prim iterate_until (fun [size] (p : Nat) => (if ((prim multiply p 2 : Nat) <? size) then (ctor Option.some [Nat] (prim multiply p 2 : Nat)) else (ctor Option.none [Nat]))) 64 1 : (Nat * Bool))),
  executable: true
}

definition{
  name: treeNodes,
  parameters: [trace : Blake3Uor.Blake3::Trace],
  result: (List Blake3Uor.Blake3::Node),
  body: (prim list_fold (fun (acc : (List Blake3Uor.Blake3::Node)) (event : Blake3Uor.Blake3::Event) => (match event

    | Blake3Uor.Blake3::Event.pushed e depth => acc

definition{
  name: covers,

  result: Bool,
  body: (if (span =? 1) then (((project n role) =? 0) && (((project n origin) =? origin) && (if (prim equal (prim bit_and (project n flags) (Blake3Uor.Blake3::flagChunkEnd) : UInt32) (0 : UInt32) : Bool) then false else true))) else (((project n role) =? 1) && (((project n origin) =? origin) && ((project n span) =? span)))),
  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: descend,
  parameters: [
    nodes : (List Blake3Uor.Blake3::Node),
    chunk : Nat,
    st : (Nat * (Nat * (List Bytes)))
  ],
  result: (Option (Nat * (Nat * (List Bytes)))),
  body: (let lo : Nat := (first st); (let size : Nat := (first (second st)); (let acc : (List Bytes) := (second (second st)); (if (size <=? 1) then (ctor Option.none [(Nat * (Nat * (List Bytes)))]) else (let left : Nat := (leftSpan size); (let node : Bytes := (prim append (chainBytes (subtreeChain nodes lo left)) (chainBytes (subtreeChain nodes (lo + left) (prim subtract size left : Nat))) : Bytes); (if (chunk <? (lo + left)) then (ctor Option.some [(Nat * (Nat * (List Bytes)))] (lo, (left, (cons node acc)))) else (ctor Option.some [(Nat * (Nat * (List Bytes)))] ((lo + left), ((prim subtract size left : Nat), (cons node acc))))))))))),
  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: sliceOf,
  parameters: [mode : Blake3Uor.Blake3::Mode, input : Bytes, chunk : Nat],
  result: (Result Bytes Blake3Uor.Blake3::Blake3Error),
  body: (match (Blake3Uor.Blake3::traceOf mode (cons input (nil Bytes)) 32)
    | Result.ok trace => (ctor Result.ok [Bytes, Blake3Uor.Blake3::Blake3Error] (let len : Nat := (prim length input : Nat); (let n : Nat := (chunkCount len); (let nodes : (List Blake3Uor.Blake3::Node) := (treeNodes trace); (if (chunk <? n) then (let parents : (List Bytes) := (second (second (first (prim iterate_until (fun [chunk, nodes] (st : (Nat * (Nat * (List Bytes)))) => (descend nodes chunk st)) 64 (0, (n, (nil Bytes))) : ((Nat * (Nat * (List Bytes))) * Bool))))); (let data : Bytes := (match (prim slice input (prim multiply chunk 1024 : Nat) (chunkSize len chunk) : (Option Bytes))
      | Option.some d => d

    | Result.error refused => (ctor Result.error [Bytes, Blake3Uor.Blake3::Blake3Error] refused)),
  axioms: [Quot.sound, propext],
  executable: true,

  name: checkLevel,
  parameters: [key : Blake3Uor.Blake3::Cv, base : UInt32, slice : Bytes, chunk : Nat, w : Walk],
  result: (Option Walk),
  body: (if (project w failed) then (ctor Option.none [Walk]) else (if ((project w size) <=? 1) then (ctor Option.none [Walk]) else (if (((project w pos) + 64) <=? definition{
  name: checkWith,
  parameters: [
    key : Blake3Uor.Blake3::Cv,
    base : UInt32,
    slice : Bytes,
    chunk : Nat,
    len : Nat,
    root : Bytes
  ],
  result: (Result Bytes SliceError),
  body: (if (chunk <? (chunkCount len)) then (if ((prim length root : Nat) =? 32) then (if ((8 <=? (prim length slice : Nat)) && ((prim equal (Blake3Uor.Blake3::blockWord slice 8 0) (Blake3Uor.Blake3::counterLow len) : Bool) && (prim equal (Blake3Uor.Blake3::blockWord slice 8 4) (Blake3Uor.Blake3::counterHigh len) : Bool))) then (let walked : Walk := (first (prim iterate_until (fun [base, chunk, key, slice] (w : Walk) => (checkLevel key base slice chunk w)) 64 (record Walk {lo := 0, size := (chunkCount len), pos := 8, expected := (Blake3Uor.Blake3::keyWordsOf root), top := true, level := 0, failed := false}) : (Walk * Bool))); (if (project walked failed) then (ctor Result.error [Bytes, SliceError] (ctor SliceError.parent (project walked level))) else (let size : Nat := (chunkSize len chunk); (let data : Bytes := (match (prim slice slice (project walked pos) (prim subtract (prim length slice : Nat) (project walked pos) : Nat) : (Option Bytes))
    | Option.some d => d
    | Option.none => 0x); (if ((prim length data : Nat) =? size) then (let pending : Blake3Uor.Blake3::Pending := (first (Blake3Uor.Blake3::chunkPending key base data 0 size chunk (nil Blake3Uor.Blake3::Event))); (let node : Blake3Uor.Blake3::Node := (Blake3Uor.Blake3::run (if (project walked top) then (Blake3Uor.Blake3::rootPending pending 0) else pending)); (if (chainsSame (Blake3Uor.Blake3::cvOf (project node output)) (project walked expected)) then (ctor Result.ok [Bytes, SliceError] data) else (ctor Result.error [Bytes, SliceError] (ctor SliceError.chunk))))) else (ctor Result.error [Bytes, SliceError] (ctor SliceError.size))))))) else (ctor Result.error [Bytes, SliceError] (ctor SliceError.header))) else (ctor Result.error [Bytes, SliceError] (ctor SliceError.rootLength))) else (ctor Result.error [Bytes, SliceError] (ctor SliceError.outOfRange))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: sliceCheck,
  parameters: [mode : Blake3Uor.Blake3::Mode, slice : Bytes, chunk : Nat, len : Nat, root : Bytes],
  result: (Result Bytes SliceError),
  body: (match (Blake3Uor.Blake3::start mode)
    | Result.ok h => (checkWith (project h keyWords) (project h modeFlags) slice chunk len root)
    | Result.error refused => (match refused
      | Blake3Uor.Blake3::Blake3Error.keyLength n => (ctor Result.error [Bytes, SliceError] (ctor SliceError.keyLength n)))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow], targets: [rust-std]}
