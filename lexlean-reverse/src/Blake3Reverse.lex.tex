definition{
  name: sub32,
  parameters: [left : UInt32, right : UInt32],
  result: UInt32,
  body: (match (prim checked_add (Blake3Uor.Blake3::widen left) (4294967296 : UInt64) : (Option UInt64))
    | Option.some big => (match (prim checked_subtract big (Blake3Uor.Blake3::widen right) : (Option UInt64))
      | Option.some diff => (Blake3Uor.Blake3::narrow diff)
      | Option.none => (0 : UInt32))
    | Option.none => (0 : UInt32)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: rotl,
  parameters: [value : UInt32, amount : UInt32],

definition{
  name: undo,
  parameters: [kind : Blake3Uor.Blake3::Op, result : UInt32, right : UInt32],
  result: UInt32,
  body: (match kind
    | Blake3Uor.Blake3::Op.opAdd => (sub32 result right)
    | Blake3Uor.Blake3::Op.opAddMsg => (sub32 result right)
    | Blake3Uor.Blake3::Op.opXor => (Blake3Uor.Blake3::xor result right)
    | Blake3Uor.Blake3::Op.opRor => (rotl result right)
    | Blake3Uor.Blake3::Op.opXorCv => (Blake3Uor.Blake3::xor result right)),
  axioms: [Quot.sound, propext],
  executable: true
}
